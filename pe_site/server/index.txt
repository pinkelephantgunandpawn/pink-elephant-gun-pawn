}
function decodeImageDataUri(v){
  const m=String(v||'').match(/^data:image\/(jpeg|jpg|png|webp);base64,(.+)$/is);
  if(!m)return null;
  const type=m[1].toLowerCase()==='jpg'?'jpeg':m[1].toLowerCase();
  return {body:Buffer.from(m[2],'base64'),contentType:`image/${type}`,ext:type==='jpeg'?'jpg':type};
}
async function storeImageRef(v,prefix='inventory'){
  if(v==null||v==='')return v||null;
  if(/^https?:\/\//i.test(v))return v;
  const decoded=decodeImageDataUri(v);
  if(!decoded)throw Object.assign(new Error('Invalid image data'),{status:400});
  const {client,cfg}=r2Client();
  const key=`${prefix}/${new Date().toISOString().slice(0,10)}/${randomUUID()}.${decoded.ext}`;
  await client.send(new PutObjectCommand({Bucket:cfg.bucket,Key:key,Body:decoded.body,ContentType:decoded.contentType,CacheControl:'public, max-age=31536000, immutable'}));
  return `${cfg.publicBase}/${key}`;
}
async function normalizeInventoryImages(x){
  const out={...x};
  if(out.image_url)out.image_url=await storeImageRef(out.image_url,'inventory');
  if(Array.isArray(out.image_urls)){
    const urls=[];
    for(const v of out.image_urls)urls.push(await storeImageRef(v,'inventory'));
    out.image_urls=urls;
    if(!out.image_url&&urls[0])out.image_url=urls[0];
  }
  return out;
}

function taxConfig(){
  const taxableStates=String(process.env.TAXABLE_STATES||'KY').split(',').map(x=>x.trim().toUpperCase()).filter(Boolean);
  const storeState=String(process.env.STORE_STATE||'KY').trim().toUpperCase();
  return {taxableStates,storeState};
}
function taxRateBps(state){
  const st=String(state||'').toUpperCase();
  const specific=Number(process.env['TAX_RATE_BPS_'+st]);
  if(Number.isFinite(specific)&&specific>=0)return specific;
  const generic=Number(process.env.SALES_TAX_RATE_BPS||600);
  return Number.isFinite(generic)&&generic>=0?generic:0;
}
function calculateSalesTax({subtotalCents,shippingCents=0,state,fulfillment}){
  const cfg=taxConfig();
  const st=(fulfillment==='pickup'?cfg.storeState:String(state||'').toUpperCase());
  if(!cfg.taxableStates.includes(st))return {tax_cents:0,state:st,rate_bps:0,taxable:false};
  const rate=taxRateBps(st);
  // Kentucky includes seller-responsible delivery charges in taxable sales price. Other states can be configured separately later.
  const taxableBase=Math.max(0,Number(subtotalCents)||0)+Math.max(0,Number(shippingCents)||0);
  return {tax_cents:Math.round(taxableBase*rate/10000),state:st,rate_bps:rate,taxable:true};
}

  const cat=String(inv.category||'').toLowerCase();

  if(/jewel|collect|accessor|coin|card/.test(cat))key='small';
  else if(/gaming|console|electronic/.test(cat))key='console';
  else if(/music|guitar|instrument/.test(cat))key='guitar';

  if(Number.isFinite(saved[0])&&saved[0]>0)p.weight=saved[0];
  return {...p,estimated:true,profile:key};
}

app.post('/api/admin/migrate-images-to-r2',auth,requireRole('admin'),async(req,res)=>{
  if(!r2Config())return res.status(503).json({error:'Configure R2 environment variables on Render first.'});
  const result={inventory_images:0,batch_images:0,items_changed:0,errors:[]};
  const inv=(await pool.query('SELECT id,image_url,image_urls FROM inventory ORDER BY created_at')).rows;
  for(const row of inv){
    try{
      let changed=false;
      let main=row.image_url;
      let gallery=Array.isArray(row.image_urls)?row.image_urls:[];
      if(typeof main==='string'&&main.startsWith('data:image/')){main=await storeImageRef(main,'inventory');result.inventory_images++;changed=true}
      const newGallery=[];
      for(const v of gallery){
        if(typeof v==='string'&&v.startsWith('data:image/')){newGallery.push(await storeImageRef(v,'inventory'));result.inventory_images++;changed=true}
        else newGallery.push(v);
      }
      if(changed){await pool.query('UPDATE inventory SET image_url=$1,image_urls=$2,updated_at=now() WHERE id=$3',[main,JSON.stringify(newGallery),row.id]);result.items_changed++}
    }catch(e){result.errors.push({type:'inventory',id:row.id,error:e.message})}
  }
  const batch=(await pool.query("SELECT id,image_data FROM batch_items WHERE image_data LIKE 'data:image/%' ORDER BY created_at")).rows;
  for(const row of batch){
    try{
      const url=await storeImageRef(row.image_data,'batch-migration');
      await pool.query('UPDATE batch_items SET image_data=$1,updated_at=now() WHERE id=$2',[url,row.id]);
      result.batch_images++;result.items_changed++;
    }catch(e){result.errors.push({type:'batch',id:row.id,error:e.message})}
  }
  await audit(req,'MIGRATE','images',null,result);
  res.json(result);
});
